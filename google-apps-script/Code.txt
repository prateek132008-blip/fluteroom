/**
 * THE FLUTE ROOM — Ebook Google Apps Script backend
 * ---------------------------------------------------------------
 * Deploy this as its OWN, SEPARATE Apps Script Web App (its own Sheet, its
 * own /exec URL) — set the resulting URL as EBOOK_CONFIG.EBOOK_GOOGLE_SCRIPT_URL
 * in js/config.js. This is completely separate from Code.gs (the flute-class
 * enrollment backend) so nothing about the class data flow is touched.
 *
 * Handles:
 *   1. doGet   → called from js/ebook.js to save/update the customer's order
 *                row BEFORE payment ("Pending") and again AFTER payment
 *                ("Paid") — upserts by Order ID so no duplicate row is ever
 *                created for the same purchase attempt. Also sends the
 *                delivery email as a fast path.
 *   2. doPost  → Razorpay webhook receiver (server-verified source of truth
 *                for "payment succeeded"), which sends the Meta Conversions
 *                API Purchase event AND the eBook delivery email, since this
 *                is the only path guaranteed to run regardless of what
 *                happens in the customer's browser.
 *
 * SECRETS: read from THIS script's own Script Properties (Project Settings →
 * Script Properties) — separate from Code.gs's properties, since this is a
 * separate Apps Script project:
 *   RAZORPAY_WEBHOOK_SECRET   (same Razorpay account/webhook secret as the rest of the site)
 *   META_PIXEL_ID             (same Pixel ID as the front end, "27629504953321020")
 *   META_CAPI_ACCESS_TOKEN    (Meta Conversions API access token — NEVER put this in front-end code)
 * ---------------------------------------------------------------
 *
 * CHANGES IN THIS VERSION (Meta Purchase Event Match Quality fixes only —
 * business logic, sheet columns, prices, IDs, and email content untouched):
 *
 *   1. sendPurchaseToMetaCAPI() now also sends "ln" (last name) — the row
 *      already has the customer's full name ("Full Name" column), but only
 *      the first name was ever split out and hashed. Last name is a real
 *      Meta match-quality parameter and this data was already being
 *      collected; it just wasn't being sent.
 *   2. sendPurchaseToMetaCAPI() now also sends "external_id" — a hash of the
 *      customer's normalized email, reused from data already collected on
 *      the eBook form (nothing new is gathered). This gives Meta an
 *      additional, independent match key on top of em/ph.
 *   3. THE ACTUAL BUG: the call to UrlFetchApp.fetch() used
 *      muteHttpExceptions: true but never read the response at all — so if
 *      Meta rejected an event (bad token, malformed hash, etc.) the failure
 *      was completely invisible, forever. The response status + body are now
 *      captured and logged (never the access token, which only ever appears
 *      inside the request URL, not the response).
 * Everything else — the delivery-email logic, the "captured"-only payment
 * check, the dedup guards, the sheet columns — is unchanged from the
 * previously-fixed version of this file.
 * ---------------------------------------------------------------
 */

const SHEET_NAME = "Ebook Orders";

// Column order — must match the header row created by setupSheet().
const COLUMNS = [
  "Timestamp", "Full Name", "Email", "WhatsApp", "Order ID",
  "Payment ID", "Payment Status", "Amount", "Product Name",
  "FBP", "FBC", "User Agent"
];

/**
 * One-time setup: run this function manually from the Apps Script editor
 * to create the sheet with the correct header row.
 */
function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  sheet.clear();
  sheet.appendRow(COLUMNS);
  sheet.setFrozenRows(1);
}

/**
 * GET requests from the eBook page (pre-payment save + post-payment update).
 * Upserts by Order ID — the SAME Order ID is generated once on the front end
 * before payment and reused for the post-payment update, so this always
 * updates the existing "Pending" row to "Paid" instead of creating a
 * duplicate row for the same purchase attempt.
 */
function doGet(e) {
  try {
    const p = e.parameter;
    const sheet = getSheet();

    const orderId = (p.orderId || "").trim();
    const email = (p.email || "").trim().toLowerCase();
    const whatsapp = (p.whatsapp || "").trim();

    const existingRow = findExistingRow(sheet, orderId, email, whatsapp);

    const rowData = [
      new Date(),
      p.fullName || "",
      p.email || "",
      p.whatsapp || "",
      orderId,
      p.paymentId || "",
      p.paymentStatus || "Pending",
      p.amount || "",
      p.productName || "",
      p.fbp || "",
      p.fbc || "",
      p.userAgent || ""
    ];

    if (existingRow > 0) {
      sheet.getRange(existingRow, 1, 1, COLUMNS.length).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }

    // Fast-path delivery email. This is a convenience backup for the webhook
    // below — dedup'd on payment ID so the customer only ever gets one email
    // no matter which path (this one or the webhook) fires first.
    if ((p.paymentStatus || "").toLowerCase() === "paid" && p.email && p.paymentId) {
      if (!wasEmailAlreadySent(p.paymentId)) {
        sendConfirmationEmail({
          fullName: p.fullName || "",
          email: p.email,
          productName: p.productName || "30 Alankaras for Flute",
          orderId: orderId,
          ebookLink: p.ebookLink || ""
        });
        markEmailSent(p.paymentId);
      }
    }

    return jsonResponse({ status: "ok" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.message });
  }
}

/**
 * POST requests from Razorpay's webhook (configured in the Razorpay
 * Dashboard → Webhooks, pointed at THIS script's /exec URL). This is the
 * authoritative, server-verified confirmation of payment, and also the
 * reliable source of the delivery email.
 */
function doPost(e) {
  try {
    const secret = PropertiesService.getScriptProperties().getProperty("RAZORPAY_WEBHOOK_SECRET");
    const signature = e.parameter["X-Razorpay-Signature"] || (e.headers && e.headers["X-Razorpay-Signature"]);
    const body = e.postData.contents;

    if (secret && signature) {
      const expected = Utilities.computeHmacSha256Signature(body, secret)
        .map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, "0")).join("");
      if (expected !== signature) {
        return jsonResponse({ status: "error", message: "Invalid webhook signature" });
      }
    }

    const payload = JSON.parse(body);
    const payment = payload.payload && payload.payload.payment && payload.payload.payment.entity;
    if (!payment) return jsonResponse({ status: "ignored" });

    // Only "captured" means the money actually settled. Razorpay also sends
    // webhook events for authorized/failed/refunded payments, and all of
    // those also contain a payment.entity — without this check, a FAILED
    // payment could get marked "Paid" and the eBook could get emailed out
    // for a payment that never actually succeeded.
    if (payment.status !== "captured") {
      return jsonResponse({ status: "ignored", reason: "payment not captured" });
    }

    // Only act on eBook payments — the same Razorpay account/webhook may also
    // receive class-enrollment webhook calls if both are ever pointed at the
    // same endpoint; this script only touches "product": "ebook" notes.
    const notes = payment.notes || {};
    if (notes.product && notes.product !== "ebook") {
      return jsonResponse({ status: "ignored" });
    }

    const sheet = getSheet();
    const orderId = notes.order_id || "";
    const email = (payment.email || "").trim().toLowerCase();
    const row = findExistingRow(sheet, orderId, email, payment.contact || "");

    let fbp = notes.fbp || "";
    let fbc = notes.fbc || "";
    let userAgent = "";
    let fullName = "";
    let productName = notes.product_name || "";
    let rowEmail = email;

    if (row > 0) {
      sheet.getRange(row, COLUMNS.indexOf("Payment ID") + 1).setValue(payment.id);
      sheet.getRange(row, COLUMNS.indexOf("Payment Status") + 1).setValue("Paid (Webhook Verified)");
      const rowValues = sheet.getRange(row, 1, 1, COLUMNS.length).getValues()[0];
      fullName = rowValues[COLUMNS.indexOf("Full Name")] || "";
      if (!productName) productName = rowValues[COLUMNS.indexOf("Product Name")] || "";
      if (!rowEmail) rowEmail = rowValues[COLUMNS.indexOf("Email")] || "";
      if (!fbp) fbp = rowValues[COLUMNS.indexOf("FBP")] || "";
      if (!fbc) fbc = rowValues[COLUMNS.indexOf("FBC")] || "";
      userAgent = rowValues[COLUMNS.indexOf("User Agent")] || "";
    }

    // This is the reliable delivery-email path — it runs regardless of
    // whether the customer's browser ever calls doGet at all. Dedup'd on
    // payment ID so this can never send a second email if doGet already did.
    if (rowEmail && !wasEmailAlreadySent(payment.id)) {
      sendConfirmationEmail({
        fullName: fullName,
        email: rowEmail,
        productName: productName || "30 Alankaras for Flute",
        orderId: orderId,
        // The Drive link travels with the payment via Razorpay "notes"
        // (see js/ebook.js), so it doesn't need to be hardcoded here too.
        ebookLink: notes.ebook_link || ""
      });
      markEmailSent(payment.id);
    }

    // NEW: split the already-available full name so BOTH first and last name
    // can be sent to Meta (previously only first name was sent).
    const nameParts = (fullName || "").trim().split(/\s+/);
    const firstName = nameParts[0] || "";
    const lastName = nameParts.slice(1).join(" ") || "";

    // ---- Meta Conversions API — server-side Purchase event ----
    // Best-effort: never let a Meta/network hiccup break webhook processing.
    try {
      sendPurchaseToMetaCAPI({
        eventId: "ebook_purchase_" + payment.id, // matches the browser Pixel's eventID exactly
        email: email,
        phone: payment.contact || "",
        firstName: firstName,
        lastName: lastName,
        value: (payment.amount || 0) / 100,
        currency: (payment.currency || "INR").toUpperCase(),
        contentName: productName,
        fbp: fbp,
        fbc: fbc,
        userAgent: userAgent
      });
    } catch (capiErr) {
      console.error("Meta CAPI send failed:", capiErr);
    }

    return jsonResponse({ status: "ok" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.message });
  }
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error(`Sheet "${SHEET_NAME}" not found. Run setupSheet() first.`);
  return sheet;
}

/**
 * Finds the row for this purchase attempt. Order ID is the primary key
 * (unique per attempt, generated before payment); email+whatsapp is kept
 * only as a fallback for backward compatibility.
 */
function findExistingRow(sheet, orderId, email, whatsapp) {
  const data = sheet.getDataRange().getValues();
  if (orderId) {
    for (let i = data.length - 1; i >= 1; i--) {
      if (String(data[i][COLUMNS.indexOf("Order ID")]).trim() === orderId) return i + 1;
    }
    return -1;
  }
  for (let i = data.length - 1; i >= 1; i--) {
    const rowEmail = String(data[i][COLUMNS.indexOf("Email")]).trim().toLowerCase();
    const rowWhatsapp = String(data[i][COLUMNS.indexOf("WhatsApp")]).trim();
    if (rowEmail === email && (whatsapp === "" || rowWhatsapp === whatsapp)) return i + 1;
  }
  return -1;
}

/**
 * Sends the post-payment eBook delivery email. Matches the same visual
 * style/tone as the class-enrollment email in Code.gs (same orange accent,
 * same structure). Unchanged from the previously-fixed version of this file.
 */
function sendConfirmationEmail(p) {
  const subject = `Your eBook is here — ${p.productName}`;
  const linkHtml = p.ebookLink
    ? `<a href="${p.ebookLink}" style="display:inline-block;background:#FF7A00;color:#fff;text-decoration:none;padding:12px 22px;border-radius:100px;font-weight:bold;margin:14px 0;">Download Your eBook</a>`
    : `<p style="color:#C0392B;">Your access link is being finalized — our team will send it separately shortly, or you can reach us on WhatsApp with your Order ID below.</p>`;

  const html = `
    <div style="font-family:Arial,sans-serif;max-width:520px;margin:0 auto;">
      <h2 style="color:#FF7A00;">Thank you, ${p.fullName}!</h2>
      <p>Your payment for <strong>${p.productName}</strong> is confirmed, and it's ready for you.</p>
      <p>${linkHtml}</p>
      <table style="width:100%;border-collapse:collapse;margin:16px 0;">
        <tr><td style="padding:8px;border-bottom:1px solid #eee;"><strong>Product</strong></td><td style="padding:8px;border-bottom:1px solid #eee;">${p.productName}</td></tr>
        <tr><td style="padding:8px;"><strong>Order ID</strong></td><td style="padding:8px;">${p.orderId}</td></tr>
      </table>
      <p><strong>Access:</strong> Lifetime — save this email or the link above for later.</p>
      <p>Questions or trouble downloading? Email us at <a href="mailto:prateek132008@gmail.com">prateek132008@gmail.com</a> or message us on WhatsApp.</p>
      <p style="color:#888;font-size:12px;margin-top:24px;">The Flute Room</p>
    </div>
  `;
  MailApp.sendEmail({
    to: p.email,
    subject: subject,
    htmlBody: html
  });
}

/**
 * Small helpers so the same payment can never cause two delivery emails to
 * go out, whether it's doGet, doPost, or both that end up triggering it for
 * the same purchase. Uses CacheService rather than the Sheet, so no sheet
 * columns/structure change. The 6-hour window comfortably covers realistic
 * webhook-retry timeframes.
 */
function wasEmailAlreadySent(paymentId) {
  if (!paymentId) return false;
  return CacheService.getScriptCache().get("emailed_" + paymentId) === "1";
}
function markEmailSent(paymentId) {
  if (!paymentId) return;
  CacheService.getScriptCache().put("emailed_" + paymentId, "1", 21600);
}

/**
 * Sends a server-side Purchase event to the Meta Conversions API.
 * Reads the Pixel ID + access token from Script Properties — NEVER hardcode
 * the access token here or anywhere in front-end code. Email/phone/name are
 * normalized and SHA-256 hashed exactly as Meta requires before being sent;
 * the plain values never leave this function.
 */
function sendPurchaseToMetaCAPI(opts) {
  const props = PropertiesService.getScriptProperties();
  const pixelId = props.getProperty("META_PIXEL_ID");
  const accessToken = props.getProperty("META_CAPI_ACCESS_TOKEN");
  if (!pixelId || !accessToken) {
    console.warn("META_PIXEL_ID / META_CAPI_ACCESS_TOKEN not set in Script Properties — skipping Conversions API call.");
    return;
  }

  const normalizedEmail = normalizeEmail(opts.email);

  const userData = {};
  if (opts.email) userData.em = [sha256Hex(normalizedEmail)];
  if (opts.phone) userData.ph = [sha256Hex(normalizePhone(opts.phone))];
  if (opts.firstName) userData.fn = [sha256Hex(opts.firstName.trim().toLowerCase())];
  // NEW: last name — same data already sitting in the "Full Name" column,
  // just never split + sent before.
  if (opts.lastName) userData.ln = [sha256Hex(opts.lastName.trim().toLowerCase())];
  // NEW: external_id — a hash of the same normalized email used for "em".
  // This is a distinct Meta match key from em/ph and is a standard way to
  // raise Event Match Quality; it uses only data already collected on the
  // eBook form, nothing new.
  if (opts.email) userData.external_id = [sha256Hex(normalizedEmail)];
  if (opts.fbp) userData.fbp = opts.fbp;
  if (opts.fbc) userData.fbc = opts.fbc;
  if (opts.userAgent) userData.client_user_agent = opts.userAgent;

  const eventPayload = {
    data: [{
      event_name: "Purchase",
      event_time: Math.floor(Date.now() / 1000),
      event_id: opts.eventId, // must match the browser Pixel eventID for deduplication
      action_source: "website",
      event_source_url: "https://theflutroom.com/alankaars-ebook.html",
      user_data: userData,
      custom_data: {
        value: opts.value,
        currency: opts.currency || "INR",
        content_name: opts.contentName || "",
        content_type: "product"
      }
    }]
  };

  const url = `https://graph.facebook.com/v19.0/${pixelId}/events?access_token=${encodeURIComponent(accessToken)}`;
  const response = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(eventPayload),
    muteHttpExceptions: true
  });

  // FIX: the response was previously fetched but never actually inspected —
  // muteHttpExceptions:true means Apps Script will not throw on a non-2xx
  // response, so a rejected/erroring Meta event was completely silent. Log
  // the status + response body (never the access token — that only appears
  // in the request URL, not the response) so a bad hash, invalid token, or
  // malformed payload actually shows up in the Apps Script execution log.
  const responseCode = response.getResponseCode();
  const responseBody = response.getContentText();
  if (responseCode < 200 || responseCode >= 300) {
    console.error(`Meta CAPI Purchase event REJECTED (HTTP ${responseCode}) for event_id=${opts.eventId}: ${responseBody}`);
  } else {
    console.log(`Meta CAPI Purchase event accepted (HTTP ${responseCode}) for event_id=${opts.eventId}: ${responseBody}`);
  }
}

function sha256Hex(input) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8);
  return digest.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, "0")).join("");
}

function normalizeEmail(email) {
  return (email || "").trim().toLowerCase();
}

// Assumes India (+91) for any 10-digit number — adjust if selling internationally.
function normalizePhone(phone) {
  const digits = (phone || "").replace(/\D/g, "");
  if (digits.length === 10) return "91" + digits;
  return digits;
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
