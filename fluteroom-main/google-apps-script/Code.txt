/**
 * THE FLUTE ROOM — Google Apps Script backend
 * ---------------------------------------------------------------
 * Handles:
 *   1. doGet   → called from the front-end (js/main.js) to save/update
 *                enrollment rows in Google Sheets (pending + paid states)
 *   2. doPost  → Razorpay webhook receiver, verifies the webhook signature
 *                server-side, marks payment as verified, and sends the
 *                confirmation email (this is the SOURCE OF TRUTH for
 *                "payment succeeded" — never trust the browser alone)
 *
 * SETUP: see README.md → "GOOGLE APPS SCRIPT DEPLOYMENT GUIDE"
 *
 * SECRETS: Key Secret and Webhook Secret are NEVER hardcoded here.
 * They are read from Script Properties (Project Settings → Script Properties):
 *   RAZORPAY_KEY_SECRET
 *   RAZORPAY_WEBHOOK_SECRET
 * ---------------------------------------------------------------
 */

const SHEET_NAME = "Enrollments";

// Column order — must match the header row created by setupSheet()
const COLUMNS = [
  "Timestamp", "Name", "Email", "WhatsApp", "Age", "Playing Level",
  "Preferred Time", "Start Date", "Plan", "Payment ID", "Payment Status", "Student Code"
];

/**
 * One-time setup: run this function manually from the Apps Script editor
 * to create the sheet with the correct header row.
 */
function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  sheet.clear();
  sheet.appendRow(COLUMNS);
  sheet.setFrozenRows(1);
}

/**
 * GET requests from the website (enrollment form + post-payment update).
 * Using GET with query params avoids the CORS/JSON POST issues that Apps
 * Script Web Apps commonly run into with fetch() from a browser.
 */
function doGet(e) {
  try {
    const p = e.parameter;
    const sheet = getSheet();

    const email = (p.email || "").trim().toLowerCase();
    const whatsapp = (p.whatsapp || "").trim();

    const existingRow = findPendingRow(sheet, email, whatsapp);

    const rowData = [
      new Date(),
      p.fullName || "",
      p.email || "",
      p.whatsapp || "",
      p.age || "",
      p.level || "",
      p.slot || "",
      p.startDate || "",
      p.plan || "",
      p.paymentId || "",
      p.paymentStatus || "Pending",
      p.studentCode || ""
    ];

    if (existingRow > 0) {
      // Update the existing pending row (e.g. Pending -> Paid) rather than
      // creating a duplicate entry for the same enrollment attempt.
      sheet.getRange(existingRow, 1, 1, COLUMNS.length).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }

    // If this call marks the enrollment as Paid, send the confirmation email.
    if ((p.paymentStatus || "").toLowerCase() === "paid" && p.email) {
      sendConfirmationEmail(p);
    }

    return jsonResponse({ status: "ok" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.message });
  }
}

/**
 * POST requests from Razorpay's webhook (configured in the Razorpay
 * Dashboard → Webhooks). This is the authoritative, server-verified
 * confirmation of payment — recommended in addition to the client-side
 * handler in js/main.js for production use.
 */
function doPost(e) {
  try {
    const secret = PropertiesService.getScriptProperties().getProperty("RAZORPAY_WEBHOOK_SECRET");
    const signature = e.parameter["X-Razorpay-Signature"] || (e.headers && e.headers["X-Razorpay-Signature"]);
    const body = e.postData.contents;

    if (secret && signature) {
      const expected = Utilities.computeHmacSha256Signature(body, secret)
        .map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, "0")).join("");
      if (expected !== signature) {
        return jsonResponse({ status: "error", message: "Invalid webhook signature" });
      }
    }

    const payload = JSON.parse(body);
    const payment = payload.payload && payload.payload.payment && payload.payload.payment.entity;
    if (!payment) return jsonResponse({ status: "ignored" });

    const sheet = getSheet();
    const email = (payment.email || "").trim().toLowerCase();
    const row = findPendingRow(sheet, email, payment.contact || "");
    if (row > 0) {
      sheet.getRange(row, COLUMNS.indexOf("Payment ID") + 1).setValue(payment.id);
      sheet.getRange(row, COLUMNS.indexOf("Payment Status") + 1).setValue("Paid (Webhook Verified)");
    }

    return jsonResponse({ status: "ok" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.message });
  }
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error(`Sheet "${SHEET_NAME}" not found. Run setupSheet() first.`);
  return sheet;
}

/**
 * Finds the most recent row for this email/whatsapp that is still Pending,
 * so a payment update overwrites the right row instead of appending a duplicate.
 */
function findPendingRow(sheet, email, whatsapp) {
  const data = sheet.getDataRange().getValues();
  for (let i = data.length - 1; i >= 1; i--) {
    const rowEmail = String(data[i][2]).trim().toLowerCase();
    const rowWhatsapp = String(data[i][3]).trim();
    if (rowEmail === email && (whatsapp === "" || rowWhatsapp === whatsapp)) {
      return i + 1; // 1-indexed sheet row
    }
  }
  return -1;
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
